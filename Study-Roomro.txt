<script>
const DATA_KEY = "study_room_data_v1";
const PASS_KEY = "study_room_password_hash_v1";
const STATS_KEY = "study_room_stats_v1";

let data = { subjects: [], questions: [] };
let stats = { answered: 0, correct: 0 };
let quiz = [];
let quizIndex = 0;
let quizCorrect = 0;
let quizAnswered = false;
let editingId = null;
let loggedIn = false;
let sessionTimer = null;

const $ = id => document.getElementById(id);
const uid = () => Date.now().toString(36) + Math.random().toString(36).slice(2, 9);

function load() {
  try {
    const saved = JSON.parse(localStorage.getItem(DATA_KEY));
    if (saved && Array.isArray(saved.subjects) && Array.isArray(saved.questions)) {
      data = saved;
    }
  } catch (_) {}
  try {
    const savedStats = JSON.parse(localStorage.getItem(STATS_KEY));
    if (savedStats && Number.isFinite(savedStats.answered) && Number.isFinite(savedStats.correct)) {
      stats = savedStats;
    }
  } catch (_) {}
}
function saveData() {
  try {
    localStorage.setItem(DATA_KEY, JSON.stringify(data));
    return true;
  } catch (_) {
    alert("保存できませんでした。ブラウザの保存容量や設定を確認してください。");
    return false;
  }
}
function saveStats() {
  try { localStorage.setItem(STATS_KEY, JSON.stringify(stats)); } catch (_) {}
}
function escapeHTML(value) {
  return String(value).replace(/[&<>"']/g, c => ({
    "&":"&amp;", "<":"&lt;", ">":"&gt;", '"':"&quot;", "'":"&#39;"
  })[c]);
}
function showPage(page) {
  $("homePage").classList.toggle("hidden", page !== "home");
  $("adminPage").classList.toggle("hidden", page !== "admin");
  if (page === "admin") updateLoginView();
  if (page === "home") renderHome();
}
function updateLoginView() {
  const configured = !!localStorage.getItem(PASS_KEY);
  $("setupForm").classList.toggle("hidden", configured);
  $("loginForm").classList.toggle("hidden", !configured || loggedIn);
  $("loginPanel").classList.toggle("hidden", loggedIn);
  $("adminTools").classList.toggle("hidden", !loggedIn);
  if (loggedIn) renderAdmin();
}
async function hashPassword(password) {
  if (!crypto.subtle) throw new Error("この環境では安全なパスワード設定機能を使えません。HTTPSのサイトで開いてください。");
  const bytes = new TextEncoder().encode(password);
  const hash = await crypto.subtle.digest("SHA-256", bytes);
  return Array.from(new Uint8Array(hash)).map(b => b.toString(16).padStart(2, "0")).join("");
}
$("setupForm").addEventListener("submit", async e => {
  e.preventDefault();
  const p = $("newPassword").value;
  if (p.length < 8) return;
  if (p !== $("confirmPassword").value) {
    $("loginMessage").textContent = "確認用パスワードが一致しません。";
    return;
  }
  try {
    localStorage.setItem(PASS_KEY, await hashPassword(p));
    loggedIn = true;
    $("loginMessage").textContent = "";
    $("setupForm").reset();
    updateLoginView();
  } catch (err) { $("loginMessage").textContent = err.message; }
});
$("loginForm").addEventListener("submit", async e => {
  e.preventDefault();
  try {
    const attempt = await hashPassword($("password").value);
    if (attempt === localStorage.getItem(PASS_KEY)) {
      loggedIn = true;
      $("loginMessage").textContent = "";
      $("loginForm").reset();
      updateLoginView();
    } else {
      $("loginMessage").textContent = "パスワードが違います。";
    }
  } catch (err) { $("loginMessage").textContent = err.message; }
});
function logout() {
  loggedIn = false;
  editingId = null;
  $("questionForm").reset();
  $("choiceFields").classList.remove("hidden");
  $("cancelEdit").classList.add("hidden");
  updateLoginView();
}
function requireAdmin() {
  if (!loggedIn) throw new Error("ログインしてください。");
}
$("subjectForm").addEventListener("submit", e => {
  e.preventDefault();
  try { requireAdmin(); } catch (err) { alert(err.message); return; }
  const name = $("subjectName").value.trim();
  if (!name) return;
  if (data.subjects.some(s => s.name.toLowerCase() === name.toLowerCase())) {
    alert("同じ名前の教科がすでにあります。");
    return;
  }
  data.subjects.push({ id: uid(), name });
  if (saveData()) {
    $("subjectForm").reset();
    renderAdmin();
    renderHome();
  }
});
function renderAdmin() {
  if (!loggedIn) return;
  const options = data.subjects.map(s => `<option value="${escapeHTML(s.id)}">${escapeHTML(s.name)}</option>`).join("");
  $("questionSubject").innerHTML = options || '<option value="">先に教科を追加してください</option>';
  const previous = $("filterSubject").value;
  $("filterSubject").innerHTML = '<option value="all">すべての教科</option>' + options;
  if (previous === "all" || data.subjects.some(s => s.id === previous)) $("filterSubject").value = previous;
  $("subjectManageList").innerHTML = data.subjects.length ? data.subjects.map(s => {
    const count = data.questions.filter(q => q.subjectId === s.id).length;
    return `<div class="item"><strong>${escapeHTML(s.name)}</strong><p class="small">${count} 問</p><button class="danger" onclick="deleteSubject('${s.id}')">教科を削除</button></div>`;
  }).join("") : '<p class="small">まだ教科がありません。</p>';
  renderQuestionList();
}
function deleteSubject(id) {
  requireAdmin();
  const s = data.subjects.find(x => x.id === id);
  if (!s) return;
  if (!confirm(`「${s.name}」と、この教科の問題をすべて削除しますか？`)) return;
  data.subjects = data.subjects.filter(x => x.id !== id);
  data.questions = data.questions.filter(q => q.subjectId !== id);
  saveData(); renderAdmin(); renderHome();
}
$("questionType").addEventListener("change", () => {
  $("choiceFields").classList.toggle("hidden", $("questionType").value !== "choice");
  $("correctText").required = $("questionType").value === "text";
});
$("questionForm").addEventListener("submit", e => {
  e.preventDefault();
  try { requireAdmin(); } catch (err) { alert(err.message); return; }
  const subjectId = $("questionSubject").value;
  if (!data.subjects.some(s => s.id === subjectId)) {
    $("questionMessage").textContent = "先に教科を追加してください。";
    return;
  }
  const type = $("questionType").value;
  const questionText = $("questionText").value.trim();
  const correctText = $("correctText").value.trim();
  const explanation = $("explanation").value.trim();
  if (!questionText || !correctText) return;
  let choices = [];
  let answer = 0;
  if (type === "choice") {
    choices = [0,1,2,3].map(i => $("choice" + i).value.trim());
    if (choices.some(c => !c)) {
      $("questionMessage").textContent = "4つの選択肢をすべて入力してください。";
      return;
    }
    answer = Number($("correctChoice").value);
    if (new Set(choices).size !== 4) {
      $("questionMessage").textContent = "選択肢はそれぞれ違う内容にしてください。";
      return;
    }
  }
  const item = { id: editingId || uid(), subjectId, type, question: questionText, choices, answer, correctText, explanation };
  if (editingId) {
    const index = data.questions.findIndex(q => q.id === editingId);
    if (index >= 0) data.questions[index] = item;
  } else {
    data.questions.push(item);
  }
  if (saveData()) {
    editingId = null;
    $("questionForm").reset();
    $("choiceFields").classList.remove("hidden");
    $("cancelEdit").classList.add("hidden");
    $("saveQuestion").textContent = "問題を追加";
    $("questionMessage").textContent = "保存しました！";
    renderAdmin(); renderHome();
  }
});
function editQuestion(id) {
  requireAdmin();
  const q = data.questions.find(x => x.id === id);
  if (!q) return;
  editingId = id;
  $("questionSubject").value = q.subjectId;
  $("questionText").value = q.question;
  $("questionType").value = q.type;
  $("choiceFields").classList.toggle("hidden", q.type !== "choice");
  [0,1,2,3].forEach(i => $("choice" + i).value = q.choices[i] || "");
  $("correctChoice").value = String(q.answer);
  $("correctText").value = q.correctText;
  $("explanation").value = q.explanation || "";
  $("saveQuestion").textContent = "変更を保存";
  $("cancelEdit").classList.remove("hidden");
  $("questionForm").scrollIntoView({ behavior: "smooth", block: "start" });
}
function cancelEdit() {
  editingId = null;
  $("questionForm").reset();
  $("choiceFields").classList.remove("hidden");
  $("cancelEdit").classList.add("hidden");
  $("saveQuestion").textContent = "問題を追加";
  $("questionMessage").textContent = "";
}
function deleteQuestion(id) {
  requireAdmin();
  if (!confirm("この問題を削除しますか？")) return;
  data.questions = data.questions.filter(q => q.id !== id);
  saveData(); renderAdmin(); renderHome();
}
$("filterSubject").addEventListener("change", renderQuestionList);
function renderQuestionList() {
  if (!loggedIn) return;
  const filter = $("filterSubject").value;
  const list = data.questions.filter(q => filter === "all" || q.subjectId === filter);
  $("questionManageList").innerHTML = list.length ? list.map(q => {
    const s = data.subjects.find(x => x.id === q.subjectId);
    return `<div class="item"><strong>${escapeHTML(s ? s.name : "未分類")}</strong><p>${escapeHTML(q.question)}</p><p class="small">形式：${q.type === "choice" ? "四択" : "記述"}｜正解：${escapeHTML(q.correctText)}</p><div class="row"><button onclick="editQuestion('${q.id}')">編集</button><button class="danger" onclick="deleteQuestion('${q.id}')">削除</button></div></div>`;
  }).join("") : '<p class="small">問題がありません。上のフォームから作成してください。</p>';
}
function renderHome() {
  const cards = $("subjectCards");
  cards.innerHTML = data.subjects.map((s,i) => {
    const colors = ["#eaf7ff","#e8f7ec","#fff6d9","#fceaf2","#efecff"];
    const count = data.questions.filter(q => q.subjectId === s.id).length;
    return `<button class="subject-card" style="background:${colors[i % colors.length]}" onclick="startQuiz('${s.id}')">${escapeHTML(s.name)}<span class="count">${count} 問　›</span></button>`;
  }).join("");
  $("emptyHome").classList.toggle("hidden", data.subjects.length > 0);
  $("statsText").textContent = `回答した問題：${stats.answered}問　／　正解：${stats.correct}問` + (stats.answered ? `　／　正答率：${Math.round(stats.correct / stats.answered * 100)}%` : "");
}
function startQuiz(subjectId) {
  const available = data.questions.filter(q => q.subjectId === subjectId);
  if (!available.length) { alert("この教科にはまだ問題がありません。"); return; }
  quiz = available.slice().sort(() => Math.random() - 0.5);
  quizIndex = 0; quizCorrect = 0;
  $("quizPanel").classList.remove("hidden");
  $("quizTitle").textContent = (data.subjects.find(s => s.id === subjectId)?.name || "勉強") + "の問題";
  $("quizPanel").scrollIntoView({ behavior: "smooth", block: "start" });
  showQuizQuestion();
}
function showQuizQuestion() {
  quizAnswered = false;
  const q = quiz[quizIndex];
  $("nextQuestion").classList.add("hidden");
  let html = `<p class="small">${quizIndex + 1} / ${quiz.length} 問</p><h3>${escapeHTML(q.question)}</h3>`;
  if (q.type === "choice") {
    html += q.choices.map((c,i) => `<button class="question-choice" data-index="${i}">${escapeHTML(c)}</button>`).join("");
  } else {
    html += '<button id="showAnswer">答えを表示する</button><div id="writtenAnswer" class="hidden item"></div>';
  }
  html += '<p id="quizFeedback" aria-live="polite"></p>';
  $("quizContent").innerHTML = html;
  if (q.type === "choice") {
    $("quizContent").querySelectorAll("[data-index]").forEach(btn => {
      btn.addEventListener("click", () => answerChoice(Number(btn.dataset.index)));
    });
  } else {
    $("showAnswer").addEventListener("click", () => {
      if (quizAnswered) return;
      quizAnswered = true;
      stats.answered++; saveStats(); renderHome();
      $("writtenAnswer").innerHTML = `<strong>正解・模範解答</strong><p>${escapeHTML(q.correctText)}</p>${q.explanation ? `<p>${escapeHTML(q.explanation)}</p>` : ""}`;
      $("writtenAnswer").classList.remove("hidden");
      $("showAnswer").disabled = true;
      $("nextQuestion").classList.remove("hidden");
    });
  }
}
function answerChoice(index) {
  if (quizAnswered) return;
  quizAnswered = true;
  const q = quiz[quizIndex];
  const isCorrect = index === q.answer;
  stats.answered++;
  if (isCorrect) { stats.correct++; quizCorrect++; quizCorrect = Math.min(quizCorrect, quiz.length); }
  saveStats(); renderHome();
  $("quizContent").querySelectorAll("[data-index]").forEach(btn => {
    btn.disabled = true;
    const i = Number(btn.dataset.index);
    if (i === q.answer) btn.classList.add("correct");
    if (i === index && !isCorrect) btn.classList.add("wrong");
  });
  $("quizFeedback").textContent = (isCorrect ? "正解！ " : "不正解。正解は「" + q.choices[q.answer] + "」。 ") + (q.explanation || "");
  $("nextQuestion").classList.remove("hidden");
}
function nextQuestion() {
  if (quizIndex + 1 < quiz.length) {
    quizIndex++; showQuizQuestion();
  } else {
    $("quizContent").innerHTML = `<h3>おつかれさま！</h3><p>${quiz.length}問中、${quizCorrect}問正解！</p><p>また挑戦してみよう。</p>`;
    $("nextQuestion").classList.add("hidden");
  }
}

function resetStats() {
  if (!confirm("学習記録をリセットしますか？")) return;
  stats = { answered: 0, correct: 0 };
  saveStats(); renderHome();
}
function exportData() {
  requireAdmin();
  const payload = { format: "study-room-backup-v1", data, stats };
  const blob = new Blob([JSON.stringify(payload,null,2)], {type:"application/json"});
  const url = URL.createObjectURL(blob);
  const a = document.createElement("a");
  a.href = url; a.download = "study-room-backup.json"; a.click();
  URL.revokeObjectURL(url);
  $("dataMessage").textContent = "バックアップファイルを作成しました。";
}
$("importFile").addEventListener("change", async e => {
  try {
    requireAdmin();
    const file = e.target.files[0];
    if (!file) return;
    if (file.size > 5 * 1024 * 1024) throw new Error("ファイルが大きすぎます。");
    const parsed = JSON.parse(await file.text());
    if (parsed.format !== "study-room-backup-v1" || !parsed.data ||
        !Array.isArray(parsed.data.subjects) || !Array.isArray(parsed.data.questions)) {
      throw new Error("対応していないバックアップファイルです。");
    }
    if (!confirm("現在の教科・問題をバックアップの内容に置き換えますか？")) return;
    const subjectIds = new Set(parsed.data.subjects.map(s => s.id));
    if (!parsed.data.subjects.every(s => typeof s.id === "string" && typeof s.name === "string") ||
        !parsed.data.questions.every(q => typeof q.id === "string" && subjectIds.has(q.subjectId) &&
        typeof q.question === "string" && ["choice","text"].includes(q.type) &&
        typeof q.correctText === "string" &&
        (q.type !== "choice" || (Array.isArray(q.choices) && q.choices.length === 4 && Number.isInteger(q.answer) && q.answer >= 0 && q.answer < 4)))) {
      throw new Error("バックアップのデータ形式が正しくありません。");
    }
    data = parsed.data;
    saveData(); renderAdmin(); renderHome();
    $("dataMessage").textContent = "読み込みました。";
  } catch (err) {
    $("dataMessage").textContent = err.message || "読み込みに失敗しました。";
  } finally { e.target.value = ""; }
});
